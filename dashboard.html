Google Sheets Cloud Backend (Percuma)

Simpan & segerakkan rekod jualan serta tetapan komisen untuk GitHub Pages.

Cara Setup Google Sheets Backend (2 Minit):
  1. Buka Google Sheets baharu di Google Drive anda.
  2. Klik menu Extensions > Apps Script.
  3. Padam kod asal, tampal kod skrip di bawah, dan klik Deploy > New deployment.
  4. Pilih jenis Web app, tetapkan Who has access kepada Anyone, kemudian salin URL Web App ke kotak di atas.
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getRange(1, 1).getValue();
  return ContentService.createTextOutput(data || '{"records":[],"payroll":{}}')
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  sheet.getRange(1, 1).setValue(e.postData.contents);
  sheet.getRange(2, 1).setValue("Last Sync: " + new Date().toISOString());
  return ContentService.createTextOutput(JSON.stringify({status: "ok"}))
    .setMimeType(ContentService.MimeType.JSON);
}
