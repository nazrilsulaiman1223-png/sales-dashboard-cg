Google Sheets Cloud Backend (Percuma)

Simpan & segerakkan rekod jualan serta tetapan komisen untuk GitHub Pages.

Kunci disimpan pada peranti ini sahaja (tidak ada dalam kod GitHub). Masukkan sekali pada setiap peranti baharu.

Belum disegerakkan.

Cara Setup Google Sheets Backend (2 Minit):
  1. Buka Google Sheets baharu, klik Extensions > Apps Script.
  2. Padam kod asal, tampal kod di bawah. Pilihan: isi SECRET dengan kata laluan anda.
  3. Deploy > New deployment > Web app, Execute as: Me, Who has access: Anyone.
  4. Jika sudah ada deployment lama: Deploy > Manage deployments > ikon pensel > Version: New version > Deploy (URL kekal sama).
  5. Data akan disegerakkan secara automatik selepas upload fail atau ubah tetapan gaji.
/**
 * Sales & Booking Dashboard - Backend v2 (Google Apps Script)
 * - Data disimpan dalam helaian "DB" (dipecah kepada bahagian kecil, tiada had 50,000 aksara)
 * - Kunci rahsia pilihan: isi SECRET di bawah supaya hanya anda boleh baca/tulis data
 */
var SECRET = '';          // contoh: 'kata-laluan-saya'  (kosong = tanpa kunci)
var SHEET_NAME = 'DB';
var CHUNK = 40000;

function getSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  return ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  if (SECRET && (!e || !e.parameter || e.parameter.key !== SECRET)) {
    return json_({ error: 'unauthorized' });
  }
  var sh = getSheet_();
  var last = sh.getLastRow();
  var text = '';
  if (last >= 1) {
    text = sh.getRange(1, 1, last, 1).getValues()
      .map(function (r) { return r[0]; }).join('');
  }
  if (!text) {
    // Sokongan data lama (versi 1: disimpan di sel A1 helaian pertama)
    var old = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0].getRange(1, 1).getValue();
    if (typeof old === 'string' && old.charAt(0) === '{') text = old;
  }
  return ContentService.createTextOutput(text || '{"records":[],"payroll":{}}')
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var data = JSON.parse(e.postData.contents);
    if (SECRET && data.key !== SECRET) return json_({ error: 'unauthorized' });
    delete data.key;

    var text = JSON.stringify(data);
    var rows = [];
    for (var i = 0; i < text.length; i += CHUNK) {
      rows.push([text.substring(i, i + CHUNK)]);
    }

    var sh = getSheet_();
    sh.clear();
    var range = sh.getRange(1, 1, rows.length, 1);
    range.setNumberFormat('@');            // paksa teks supaya tidak ditukar jadi nombor/formula
    range.setValues(rows);
    sh.getRange(1, 3).setValue('Last Sync: ' + new Date().toISOString());

    return json_({ status: 'ok', records: (data.records || []).length });
  } catch (err) {
    return json_({ error: String(err) });
  } finally {
    lock.releaseLock();
  }
}
